## 服务端
```cj
public struct Server<DE, EN, M, T> <: Resource where EN <: MessageCopier, M <: Protocol<EN, DE>, T <: DataFields<T> {
    /**
     * @param queueSize 数据传输任务队列
     * @param bindAt    绑定的端口号
     * @param checkDuration tcp连接有效性检查周期
     * @param checker   tcp连接有效性检查函数，返回None说明连接已失效，应立即执行关闭操作
     *                  checker函数应当维持心跳机制，维持心跳状态，如果服务端发出心跳却一直没有收到客户端的心跳响应应当判定为连接失效，此时不必再返回有效心跳数据而是应当返回None
     */
    public init(queueSize!: Int64 = 1024, bindAt!: UInt16, checkDuration!: Duration, checker!: () -> ?M)
    /**
     * 创建服务器构造器
     */
    public static func build(queueSize!: Int64 = 1024, checkDuration!: Duration, checker!: () -> ?M, bindAt!: UInt16): ServerBuilder<DE, EN, M, T>
    /**
     * 启动服务端
     * 接收到客户端请求，服务端实例自动解码，以解码的实例作为参数调用executor，将返回的消息写入socket
     */
    public func start(executor: (DE) -> M): Unit
    public func isClosed(): Bool
    public func close(): Unit
}
```
